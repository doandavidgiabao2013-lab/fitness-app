function histExDetail(e) {
  const sets=e.sets||[]; if(!sets.length) return'<span class="muted">No sets</span>';
  const groups=[]; sets.forEach(s=>{const last=groups[groups.length-1];if(last&&last.w===s.w&&last.r===s.r&&!!last.skip===!!s.skip&&!!last.warm===!!s.warm){last.n++}else groups.push({w:s.w,r:s.r,n:1,skip:!!s.skip,warm:!!s.warm})});
  const u=e.unit; const wl=(g)=>{const L=load(e,g.w);return e.bw==='bw'?'BW':e.bw==='plus'?`BW+${cv(g.w,u)}`:e.bw==='minus'?`BW−${cv(g.w,u)}`:(u==='kg'||u==='lbs')?`${cv(g.w,u)} ${u}`:'reps'};
  return groups.map(g=>`${g.n}×${g.r}${g.w>0||e.bw!=='none'?' @ '+wl(g):''}${g.warm?' 🔥':''}${g.skip?' ⏭':''}`).join(', ');
}

const nfmt = v => v == null ? '—' : (+v).toLocaleString('en-US', { maximumFractionDigits: Math.abs(v) < 10 ? 2 : 1 });
const nunit = (v, u) => v == null ? '—' : `${nfmt(v)} ${u}`;
const norm = s => String(s||'').toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g,'');
function showFoodPage(page, html, top) { openSubpage(html); ui.page = page; $('subpage').scrollTop = top || 0; }
const pageHead = (title, back, backLabel, right) => `<div class="fhead"><button class="btn btn-ghost btn-sm" data-action="${back}">← ${backLabel||'Back'}</button><div class="fhead-title">${title}</div>${right||''}</div>`;
const foodTags = x => `<span class="tag">${esc(x.cat)}</span>${x.prep?` <span class="tag ac">${esc(x.prep)}</span>`:''}`;
const byName = (a, b) => a.name.localeCompare(b.name);

// Every word typed must appear in the name, an alternative name or the raw/cooked state.
// Category names are matched from 3 letters up, so one letter doesn't match whole categories.
function searchFoods(q, list) {
  const nq = norm(q).trim(), words = nq.split(/\s+/).filter(Boolean);
  if (!words.length) return list.slice().sort(byName);
  const rank = x => { const n = norm(x.name); return n.startsWith(nq) ? 0 : n.split(/\W+/).some(w=>w.startsWith(nq)) ? 1 : n.includes(nq) ? 2 : 3; };
  return list.filter(x => { const text = norm(`${x.name} ${x.aliases} ${x.prep}`), cat = norm(x.cat); return words.every(w => text.includes(w) || (w.length >= 3 && cat.includes(w))); })
    .sort((a,b) => rank(a) - rank(b) || byName(a,b));
}

    const tone=(freq,start,dur)=>{const o=ctx.createOscillator(),g=ctx.createGain();o.type='sine';o.frequency.value=freq;g.gain.value=.0001;o.connect(g);g.connect(ctx.destination);g.gain.exponentialRampToValueAtTime(.22,t+start+.01);g.gain.exponentialRampToValueAtTime(.0001,t+start+dur);o.start(t+start);o.stop(t+start+dur+.02)};

  const remaining=t.paused!=null?t.paused:Math.max(0,t.end-Date.now()), sc=Math.ceil(remaining/1000), mm=pad(Math.floor(sc/60)), ss=pad(sc%60);

  const remaining=t.end-Date.now();

function onDataChange(e) {
  const el=e.target, dc=el.dataset&&el.dataset.change; if(!dc)return;
  const u=gu();
  switch(dc){
    case 'sw': {const a=S.activeWorkout;if(!a)break;const ex=a.exs[+el.dataset.ex],s=ex.sets[+el.dataset.id],v=fk(el.value,u);if(v!=null)s.w=v;dsave();break}
    case 'sr': {const a=S.activeWorkout;if(!a)break;const exi=+el.dataset.ex,ex=a.exs[exi],s=ex.sets[+el.dataset.id];if(el.value!=='')s.r=Math.max(0,parseInt(el.value)||0);dsave();break}
    case 'srpe': {const a=S.activeWorkout;if(!a)break;const ex=a.exs[+el.dataset.ex],s=ex.sets[+el.dataset.id],raw=String(el.value).trim();if(raw===''){s.rpe=null}else{const v=parseFloat(raw.replace(',','.'));if(isFinite(v))s.rpe=Math.min(10,Math.max(0,Math.round(v*2)/2))}dsave();break}
    case 'snote': {const a=S.activeWorkout;if(!a)break;const ex=a.exs[+el.dataset.ex],s=ex.sets[+el.dataset.id];s.note=el.value.slice(0,140);dsave();break}
    case 'rname': ui.rname=el.value; break;
    case 'hval': {const x=S.habits.list.find(h=>h.id===el.dataset.id);if(!x)break;const k=ui.hd;S.habits.log[k]=S.habits.log[k]||{};if(el.value!=='')S.habits.log[k][x.id]=Math.max(0,num(el.value,0));dsave();break}
    case 'daynote': {const k=el.dataset.date;S.calendar[k]=S.calendar[k]||{note:''};S.calendar[k].note=el.value;dsave();break}
    case 'unit': S.settings.unit=el.value;save();render();break;
    case 'theme': S.settings.theme=el.value;applyTheme();save();break;
    case 'rest': if(el.value!=='')S.settings.rest=Math.min(600,Math.max(5,parseInt(el.value)||90));dsave();break;
    case 'bw': {const v=fk(el.value,u);if(v!=null){S.bodyweight[dk()]=v}dsave();break}
    case 'goal': if(el.value!=='')S.settings.goal=Math.min(14,Math.max(1,parseInt(el.value)||4));dsave();break;
    case 'goalKcal': if(el.value!=='')S.settings.macroGoals.kcal=Math.max(0,num(el.value,2200));dsave();break;
    case 'goalCarbs': if(el.value!=='')S.settings.macroGoals.carbs=Math.max(0,num(el.value,250));dsave();break;
    case 'goalFat': if(el.value!=='')S.settings.macroGoals.fat=Math.max(0,num(el.value,70));dsave();break;
    case 'goalProtein': if(el.value!=='')S.settings.macroGoals.protein=Math.max(0,num(el.value,150));dsave();break;
    case 'foodSearch': { ui.fs.q=el.value; $('fsResults').innerHTML=searchResultsHTML(); break; }
    case 'dbSearch': { ui.db.q=el.value; $('dbResults').innerHTML=dbResultsHTML(); break; }
    case 'fdtQty': { ui.fdt.qty=el.value; $('fdtCalc').innerHTML=detailCalcHTML(); if(ui.fdt.full)$('fdtFull').innerHTML=detailFullHTML(); break; }

function init() {
  applyTheme();
  const setAppH = () => {
    const h = window.visualViewport ? window.visualViewport.height : window.innerHeight;
    document.documentElement.style.setProperty('--app-h', h+'px');
  };
  setAppH();
  if(window.visualViewport){window.visualViewport.addEventListener('resize',setAppH);window.visualViewport.addEventListener('scroll',setAppH)}
  window.addEventListener('resize',setAppH);

  document.addEventListener('pointerdown',unlockAudio,{once:true});
  document.addEventListener('touchstart',unlockAudio,{once:true,passive:true});

  // Touch handling
  let tsx=0,tsy=0,tmoved=false,lastTapTs=0;
  document.addEventListener('touchstart',e=>{const t=e.touches[0];if(!t)return;tsx=t.clientX;tsy=t.clientY;tmoved=false},{passive:true});
  document.addEventListener('touchmove',e=>{const t=e.touches[0];if(!t)return;if(Math.abs(t.clientX-tsx)>10||Math.abs(t.clientY-tsy)>10)tmoved=true},{passive:true});
  document.addEventListener('touchend',e=>{
    if(tmoved)return;
    const ct=e.changedTouches&&e.changedTouches[0];if(!ct)return;
    const el=document.elementFromPoint(ct.clientX,ct.clientY);if(!el)return;
    if(el.id==='modal'){lastTapTs=Date.now();e.preventDefault();closeModal();return}
    const hit=el.closest('[data-action],[data-pick-emoji]');if(!hit||hit.disabled)return;
    lastTapTs=Date.now();e.preventDefault();handleClick({target:hit});
  },{passive:false});
  document.addEventListener('click',e=>{if(Date.now()-lastTapTs<500)return;handleClick(e)});

  // Input handling
  document.addEventListener('input',e=>{
    const t=e.target;
    if(!t||!t.dataset||!t.dataset.change)return;
    const key='_lv_'+t.dataset.change;
    if(t[key]===t.value)return;
    t[key]=t.value;
    onDataChange(e);
  });
  document.addEventListener('change',e=>{
    if(e.target&&e.target.id==='impfile'){const file=e.target.files&&e.target.files[0];if(!file)return;
      const rd=new FileReader();rd.onload=()=>{try{const obj=JSON.parse(rd.result);S=migrate(obj);save();closeModal();render();toast('Backup imported ✅')}catch(err){toast('Invalid file')}};rd.readAsText(file);return}
    if(!e.target||!e.target.dataset||!e.target.dataset.change)return;
    const t=e.target,key='_lv_'+t.dataset.change,already=t[key]===t.value;
    t[key]=t.value;if(!already)onDataChange(e);
    if(!['rmw','rmr','rname','foodSearch','dbSearch','fdtQty','fdtMeal'].includes(t.dataset.change))save();
  });
  document.addEventListener('keydown',e=>{if(e.key==='Enter'&&e.target&&e.target.id==='fsQ'){e.preventDefault();rememberSearch();e.target.blur()}});
  $('modal').addEventListener('click',e=>{if(Date.now()-lastTapTs<500)return;if(e.target.id==='modal')closeModal()});
  document.addEventListener('focusin',e=>{
    const t=e.target;if(!t||!/^(INPUT|TEXTAREA|SELECT)$/.test(t.tagName))return;
    setTimeout(()=>{try{t.scrollIntoView({block:'center',behavior:'smooth'})}catch(e){}},300);
  });
  window.addEventListener('resize',()=>{const b=$('tbar');if(b&&b.classList.contains('show'))document.documentElement.style.setProperty('--tbar-h',b.offsetHeight+2+'px')});
  const flushSave=()=>{dsave.flush();save()};
  document.addEventListener('visibilitychange',()=>{if(document.hidden)flushSave()});
  window.addEventListener('pagehide',flushSave);
  window.addEventListener('blur',flushSave);

  if(S.timer)renderTimerBar();
  render();
}

init();
